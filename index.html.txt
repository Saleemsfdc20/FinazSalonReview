const $ = s => document.querySelector(s);
const $$ = s => [...document.querySelectorAll(s)];
const pick = arr => arr[Math.floor(Math.random()*arr.length)];
const bucket = r => r>=4 ? "pos" : r===3 ? "mid" : "neg";

function hash(str){ let h=5381; for(let i=0;i<str.length;i++) h=((h<<5)+h)+str.charCodeAt(i); return (h>>>0).toString(36); }

const HIST_KEY = "rv_hist_" + hash(CONFIG.BUSINESS_NAME);
function seen(h){ try{ return JSON.parse(localStorage.getItem(HIST_KEY)||"[]").includes(h);}catch(e){return false} }
function remember(h){
  try{
    const a = JSON.parse(localStorage.getItem(HIST_KEY)||"[]"); a.push(h);
    localStorage.setItem(HIST_KEY, JSON.stringify(a.slice(-300)));
  }catch(e){}
}

function buildDraft(rating, tags){
  const b = bucket(rating);
  let text = "";
  for(let attempt=0; attempt<40; attempt++){
    const parts = [ pick(OPEN[b]).replace("{biz}", CONFIG.BUSINESS_NAME) ];
    const chosen = tags.length ? [...tags].sort(()=>Math.random()-.5).slice(0,3) : [];
    chosen.forEach(t => parts.push(pick(DETAIL[t][b])));
    if(!chosen.length) parts.push(pick(GENERIC[b]));
    parts.push(pick(CLOSE[b]));
    text = parts.join(" ");
    if(!seen(hash(text))) break;   // avoid repeating a draft this device has already received
  }
  remember(hash(text));
  return text;
}

async function copyText(txt){

  }
}

  const p = feedbackPayload();

  // store a local copy
  try{
    const k="rv_feedback_"+hash(CONFIG.BUSINESS_NAME);

  if(CONFIG.FEEDBACK_ENDPOINT){

    }catch(e){}
  }
  btn.textContent="Send feedback";

});

function finish(title,msg,celebrate){
  $("#doneTitle").textContent=title; $("#doneMsg").textContent=msg;
  const wa=$("#waBtn");

    wa.style.display="flex";

}

/* ---------- confetti ---------- */
function confetti(){
  const cv=$("#confetti"), ctx=cv.getContext("2d");
  cv.width=innerWidth; cv.height=innerHeight;
  const colors=["#4f46e5","#7c3aed","#fbbf24","#22c55e","#ef4444","#06b6d4"];
  const ps=Array.from({length:110},()=>({
    x:Math.random()*cv.width, y:-20-Math.random()*cv.height*.4,
    r:4+Math.random()*5, c:colors[Math.floor(Math.random()*colors.length)],
    vy:2+Math.random()*3.5, vx:-1.5+Math.random()*3, a:Math.random()*6
  }));
  let f=0;
  (function tick(){
    ctx.clearRect(0,0,cv.width,cv.height);
    ps.forEach(p=>{
      p.y+=p.vy; p.x+=p.vx+Math.sin((f+p.a)/10); p.a+=.1;
      ctx.fillStyle=p.c; ctx.fillRect(p.x,p.y,p.r,p.r*1.6);
    });
    if(++f<170) requestAnimationFrame(tick); else ctx.clearRect(0,0,cv.width,cv.height);
  })();
}
